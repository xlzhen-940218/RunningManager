cmake_minimum_required(VERSION 3.16)

# ---------------------------------------------------------------------------
# MSYS2 / Cygwin shell support
#
# MSYS2 provides an MSYS CMake (/usr/bin/cmake) and a MinGW-w64 CMake
# (/mingw64/bin/cmake).  The former targets the MSYS runtime and reports
# CMAKE_SYSTEM_NAME=MSYS, while the latter reports Windows.  When the MSYS
# build of CMake is used together with the MinGW-w64 compiler, the resulting
# executable is a normal Windows program, so configure for Windows instead of
# hitting the "unsupported platform" branch below.
#
# This probe has to run before project(), because the platform modules are
# selected while the project is configured.  If the compiler found in PATH is
# the MSYS/Cygwin one (target *-msys / *-cygwin), nothing is changed and the
# regular platform check below reports the problem with instructions.
# ---------------------------------------------------------------------------
if(CMAKE_HOST_SYSTEM_NAME MATCHES "^(MSYS|CYGWIN)"
   AND (NOT CMAKE_SYSTEM_NAME OR CMAKE_SYSTEM_NAME STREQUAL CMAKE_HOST_SYSTEM_NAME))
    if(DEFINED CMAKE_CXX_COMPILER)
        set(_rm_bootstrap_compiler "${CMAKE_CXX_COMPILER}")
    else()
        find_program(_rm_bootstrap_compiler NAMES c++ g++)
        set(_rm_bootstrap_compiler "${_rm_bootstrap_compiler}")
        unset(_rm_bootstrap_compiler CACHE)
    endif()

    if(_rm_bootstrap_compiler)
        execute_process(
            COMMAND "${_rm_bootstrap_compiler}" -dumpmachine
            OUTPUT_VARIABLE _rm_bootstrap_target
            OUTPUT_STRIP_TRAILING_WHITESPACE
            ERROR_QUIET
            RESULT_VARIABLE _rm_bootstrap_result
        )
        if(_rm_bootstrap_result EQUAL 0 AND _rm_bootstrap_target MATCHES "mingw|windows")
            set(CMAKE_SYSTEM_NAME Windows)
            # A build tree configured earlier with the MSYS/Cygwin platform
            # remembers that platform in CMakeSystem.cmake, which would
            # override the setting above.  It is a generated file, so drop it
            # and let CMake recreate it for the Windows platform.
            set(_rm_stale_system_file
                "${CMAKE_BINARY_DIR}/CMakeFiles/${CMAKE_VERSION}/CMakeSystem.cmake")
            if(EXISTS "${_rm_stale_system_file}")
                file(READ "${_rm_stale_system_file}" _rm_stale_system_content)
                if(_rm_stale_system_content MATCHES "set\\(CMAKE_SYSTEM_NAME [\"']?(MSYS|CYGWIN)")
                    file(REMOVE "${_rm_stale_system_file}")
                endif()
            endif()
            set(RM_MSYS_MINGW TRUE)
            message(STATUS "MSYS/Cygwin shell: building for Windows with '${_rm_bootstrap_target}'")
        endif()
    endif()
endif()

if(RM_MSYS_MINGW AND CMAKE_GENERATOR MATCHES "Ninja")
    # ninja.exe is a native Windows binary: it starts the compiler directly and
    # never converts the POSIX paths generated by the MSYS/Cygwin CMake, so its
    # compiler check fails.  The Makefile generators run commands through the
    # MSYS/Cygwin shell, where path conversion happens automatically.
    message(FATAL_ERROR
        "The Ninja generator cannot be used with the MSYS/Cygwin CMake and a native "
        "MinGW-w64 compiler.\n"
        "Use the default \"Unix Makefiles\" generator instead (omit -G Ninja).")
endif()

project(RunningManager VERSION 1.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED True)
set(CMAKE_CXX_EXTENSIONS OFF)

# Fetch Dear ImGui
include(FetchContent)
FetchContent_Declare(
    imgui
    GIT_REPOSITORY https://github.com/ocornut/imgui.git
    GIT_TAG v1.90.1
    GIT_SHALLOW TRUE
)
FetchContent_MakeAvailable(imgui)

if(CMAKE_HOST_SYSTEM_NAME MATCHES "^(MSYS|CYGWIN)" AND CMAKE_SYSTEM_NAME STREQUAL "Windows")
    # The MSYS/Cygwin build of CMake keeps paths in POSIX form.  MinGW reads
    # @response files itself and therefore bypasses the MSYS runtime's path
    # translation, so keep the include flags on the command line where they
    # are translated before the native compiler sees them.
    set(CMAKE_CXX_USE_RESPONSE_FILE_FOR_INCLUDES 0)
endif()

add_executable(RunningManager)

# Dear ImGui core
target_sources(RunningManager PRIVATE
    ${imgui_SOURCE_DIR}/imgui.cpp
    ${imgui_SOURCE_DIR}/imgui_draw.cpp
    ${imgui_SOURCE_DIR}/imgui_tables.cpp
    ${imgui_SOURCE_DIR}/imgui_widgets.cpp
)

# Shared UI
target_sources(RunningManager PRIVATE
    src/UIManager.cpp
    src/UIManager.h
)

target_include_directories(RunningManager PRIVATE
    ${imgui_SOURCE_DIR}
    ${imgui_SOURCE_DIR}/backends
    src
)

if(WIN32)
    # ---------------------------------------------------------------------
    # Windows (Win32 + Direct3D 11)
    # ---------------------------------------------------------------------
    enable_language(RC)
    set_target_properties(RunningManager PROPERTIES WIN32_EXECUTABLE TRUE)

    target_sources(RunningManager PRIVATE
        src/main.cpp
        src/ProcessManager_win.cpp
        src/ProcessManager.h
        src/SystemMonitor_win.cpp
        src/SystemMonitor.h
        src/resource.h
        src/RunningManager.rc
        ${imgui_SOURCE_DIR}/backends/imgui_impl_win32.cpp
        ${imgui_SOURCE_DIR}/backends/imgui_impl_dx11.cpp
    )

    target_compile_definitions(RunningManager PRIVATE
        UNICODE
        _UNICODE
        WIN32_LEAN_AND_MEAN
        NOMINMAX
    )

    target_link_libraries(RunningManager PRIVATE
        d3d11
        d3dcompiler
        dwmapi
        psapi
        wtsapi32
        version
        pdh
    )

    if(MINGW)
        # Avoid a run-time dependency on the MinGW runtime DLLs, so the
        # executable also works outside an MSYS2 shell.
        target_link_options(RunningManager PRIVATE -static)
    endif()
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    # ---------------------------------------------------------------------
    # Linux (SDL2 + OpenGL 3, /proc based process and system information)
    # ---------------------------------------------------------------------
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
    find_package(OpenGL REQUIRED)
    find_package(Threads REQUIRED)

    target_sources(RunningManager PRIVATE
        src/main_linux.cpp
        src/ProcessManager_linux.cpp
        src/ProcessManager.h
        src/SystemMonitor_linux.cpp
        src/SystemMonitor.h
        ${imgui_SOURCE_DIR}/backends/imgui_impl_sdl2.cpp
        ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp
    )

    target_link_libraries(RunningManager PRIVATE
        PkgConfig::SDL2
        OpenGL::GL
        Threads::Threads
    )

    # EWMH integration for "Bring to Front" and GUI-process detection.
    find_package(X11)
    if(X11_FOUND)
        target_compile_definitions(RunningManager PRIVATE RM_HAVE_X11=1)
        target_link_libraries(RunningManager PRIVATE X11::X11)
    else()
        message(STATUS "X11 headers not found: window activation will use xdotool when available")
    endif()

    target_compile_options(RunningManager PRIVATE -Wall -Wextra)

    install(TARGETS RunningManager RUNTIME DESTINATION bin)
else()
    if(CMAKE_HOST_SYSTEM_NAME MATCHES "^(MSYS|CYGWIN)")
        message(FATAL_ERROR
            "RunningManager supports Windows and Linux only, but this build targets "
            "'${CMAKE_SYSTEM_NAME}'.\n"
            "CMake was started from an MSYS/Cygwin shell, but no Windows-target "
            "compiler was detected (the selected compiler targets "
            "'${CMAKE_SYSTEM_NAME}').\n"
            "\n"
            "Use the MinGW-w64 toolchain, for example from the \"MSYS2 MINGW64\" "
            "shell:\n"
            "    cmake -S . -B build -DCMAKE_BUILD_TYPE=Release\n"
            "or call the MinGW CMake from the current shell:\n"
            "    /mingw64/bin/cmake -S . -B build -DCMAKE_BUILD_TYPE=Release\n"
            "\n"
            "Install it with: pacman -S mingw-w64-x86_64-gcc mingw-w64-x86_64-cmake\n"
            "(MSVC users: run these commands from a Visual Studio Developer "
            "PowerShell.)")
    else()
        message(FATAL_ERROR
            "RunningManager currently supports Windows and Linux only "
            "(target platform: '${CMAKE_SYSTEM_NAME}')")
    endif()
endif()
